# 案例 4：基于 SVM-PSO 模型的中小企业信用评分预测
**问题背景**：中小企业是国民经济的重要组成部分，但融资难问题一直制约其发展。银行等金融机构需要对中小企业的信用风险进行评估，以决定是否放贷及放贷额度。中小企业信用评分受企业财务状况、经营年限、行业前景、负责人信用记录等多因素影响，且样本数据相对较少。
**问题描述**：某商业银行计划构建中小企业信用评分模型，根据企业的各项指标预测其信用等级（分为 AAA、AA、A、BBB、BB、B、CCC 共 7 个等级）。要求模型能在小样本、高维特征情况下准确预测，帮助银行有效控制信贷风险。
**数据情况**：提供该银行过去 5 年的中小企业贷款客户数据，每条数据包含企业的财务指标（资产负债率、净利润率、流动比率等 20 项）、经营数据（成立年限、员工人数、年销售额等）、行业类别、负责人信用记录及最终信用等级。数据量约 2000 条，部分企业存在财务数据不完整的情况。
案例 4：SVM-PSO 模型中小企业信用评分预测代码

| import pandas as pd<br>import numpy as np<br>from sklearn.preprocessing import LabelEncoder, StandardScaler<br>from sklearn.model_selection import train_test_split<br>from sklearn.svm import SVC<br>from sklearn.metrics import classification_report, accuracy_score<br>from pyswarm import pso<br>import joblib<br># 数据加载与预处理<br>data = pd.read_csv('small_business_credit.csv')<br>data = data.dropna()<br># 信用等级编码<br>le = LabelEncoder()<br>data['credit_rating'] = le.fit_transform(data['credit_rating'])<br># 特征与目标变量<br>X = data.drop('credit_rating', axis=1)<br>y = data['credit_rating']<br># 数据标准化<br>scaler = StandardScaler()<br>X_scaled = scaler.fit_transform(X)<br># 划分训练集和测试集<br>X_train, X_test, y_train, y_test = train_test_split(X_scaled, y, test_size=0.2, random_state=42)<br># 定义SVM参数优化目标函数<br>def svm_objective(params):<br>C = 10 **params[0]  # 惩罚因子，取指数避免负数值<br>gamma = 10** params[1]  # 核函数参数<br>svm = SVC(C=C, gamma=gamma, kernel='rbf', random_state=42)<br>svm.fit(X_train, y_train)<br>y_pred = svm.predict(X_train)<br>return 1 - accuracy_score(y_train, y_pred)  # 最小化错误率<br># 参数范围 (-5, 5) 对应 (1e-5, 1e5)<br>lb = [-5, -5]<br>ub = [5, 5]<br># PSO优化<br>best_params, _ = pso(svm_objective, lb, ub, swarmsize=20, maxiter=50, debug=True)<br>best_C = 10 ** best_params[0]<br>best_gamma = 10 ** best_params[1]<br># 最优SVM模型训练与预测<br>best_svm = SVC(C=best_C, gamma=best_gamma, kernel='rbf', random_state=42)<br>best_svm.fit(X_train, y_train)<br>y_pred = best_svm.predict(X_test)<br># 评估模型<br>print(f'最优参数: C={best_C}, gamma={best_gamma}')<br>print(f'准确率: {accuracy_score(y_test, y_pred)}')<br>print(classification_report(y_test, y_pred))<br># 保存模型<br>joblib.dump(best_svm, 'svm_pso_credit.pkl')<br>joblib.dump(scaler, 'scaler_credit.pkl')<br>joblib.dump(le, 'label_encoder_credit.pkl') |
