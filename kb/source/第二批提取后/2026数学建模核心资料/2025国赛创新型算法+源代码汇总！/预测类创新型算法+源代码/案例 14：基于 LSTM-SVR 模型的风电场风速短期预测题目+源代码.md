# 案例 14：基于 LSTM-SVR 模型的风电场风速短期预测
**问题背景**：风能是一种清洁可再生能源，风电场的发电量取决于风速，而风速具有强烈的非线性和随机性，受地形、气候、湍流等因素影响，短期风速预测难度较大。准确预测风速对风电场的功率调度、电网稳定运行具有重要意义。
**问题描述**：某风电场需要对场内 10 台风机未来 48 小时的每 10 分钟风速进行预测。要求模型能够先捕捉风速的整体非线性变化趋势，再对预测残差进行修正，以降低过拟合风险，提高预测精度，为风电场的发电计划制定提供支持。
**数据情况**：提供该风电场过去 3 年的每 10 分钟风速数据，同时提供风机的位置信息、地形数据、气象数据（温度、湿度、气压、风向等）。数据存在因设备维护导致的部分时段缺失，且风速受湍流影响存在高频波动。
案例 14：LSTM-SVR 模型风电场风速短期预测代码

| import pandas as pd<br>import numpy as np<br>import matplotlib.pyplot as plt<br>from sklearn.preprocessing import MinMaxScaler<br>from sklearn.svm import SVR<br>from sklearn.metrics import mean_absolute_error, mean_squared_error<br>from tensorflow.keras.models import Sequential<br>from tensorflow.keras.layers import LSTM, Dense<br>import joblib<br># 数据加载与预处理<br>data = pd.read_csv('wind_farm_data.csv', parse_dates=['time'], index_col='time')<br>wind_speed = data['wind_speed'].values.reshape(-1, 1)<br># 数据归一化<br>scaler = MinMaxScaler(feature_range=(0, 1))<br>wind_scaled = scaler.fit_transform(wind_speed)<br># 构建序列数据<br>def create_sequences(data, look_back=12):<br>X, y = [], []<br>for i in range(len(data) - look_back - 1):<br>X.append(data[i:i+look_back, 0])<br>y.append(data[i+look_back, 0])<br>return np.array(X), np.array(y)<br>look_back = 12  # 用过去12个10分钟数据预测下一个10分钟<br>X, y = create_sequences(wind_scaled, look_back)<br># 划分训练集和测试集<br>train_size = int(len(X) * 0.8)<br>X_train, X_test = X[:train_size], X[train_size:]<br>y_train, y_test = y[:train_size], y[train_size:]<br># 重塑LSTM输入形状 [samples, time steps, features]<br>X_train_lstm = X_train.reshape(X_train.shape[0], X_train.shape[1], 1)<br>X_test_lstm = X_test.reshape(X_test.shape[0], X_test.shape[1], 1)<br># 第一步：LSTM模型预测<br>lstm_model = Sequential()<br>lstm_model.add(LSTM(64, input_shape=(look_back, 1), return_sequences=False))<br>lstm_model.add(Dense(32, activation='relu'))<br>lstm_model.add(Dense(1))<br>lstm_model.compile(loss='mse', optimizer='adam')<br>lstm_model.fit(X_train_lstm, y_train, epochs=30, batch_size=32, verbose=1)<br># LSTM预测及残差计算<br>lstm_pred_train = lstm_model.predict(X_train_lstm).flatten()<br>lstm_pred_test = lstm_model.predict(X_test_lstm).flatten()<br>train_residual = y_train - lstm_pred_train  # 训练残差<br>test_residual = y_test - lstm_pred_test    # 测试残差<br># 第二步：SVR模型拟合残差<br>svr_model = SVR(kernel='rbf', C=10, gamma=0.1)<br>svr_model.fit(X_train, train_residual)  # 用原始特征预测残差<br># SVR残差预测<br>svr_residual_pred = svr_model.predict(X_test)<br># 组合预测：LSTM预测 + SVR残差修正<br>final_pred = lstm_pred_test + svr_residual_pred<br># 反归一化<br>final_pred_actual = scaler.inverse_transform(final_pred.reshape(-1, 1))<br>y_test_actual = scaler.inverse_transform(y_test.reshape(-1, 1))<br>lstm_pred_actual = scaler.inverse_transform(lstm_pred_test.reshape(-1, 1))<br># 评估模型<br>print(f'LSTM模型MAE: {mean_absolute_error(y_test_actual, lstm_pred_actual)}')<br>print(f'LSTM-SVR组合模型MAE: {mean_absolute_error(y_test_actual, final_pred_actual)}')<br># 可视化结果<br>plt.figure(figsize=(12, 6))<br>plt.plot(y_test_actual[:100], label='实际风速')<br>plt.plot(lstm_pred_actual[:100], label='LSTM预测', alpha=0.7)<br>plt.plot(final_pred_actual[:100], label='组合模型预测', alpha=0.7)<br>plt.legend()<br>plt.savefig('wind_speed_prediction.png')<br>plt.show()<br># 保存模型<br>lstm_model.save('lstm_wind_speed.h5')<br>joblib.dump(svr_model, 'svr_residual_wind.pkl')<br>joblib.dump(scaler, 'scaler_wind.pkl') |
