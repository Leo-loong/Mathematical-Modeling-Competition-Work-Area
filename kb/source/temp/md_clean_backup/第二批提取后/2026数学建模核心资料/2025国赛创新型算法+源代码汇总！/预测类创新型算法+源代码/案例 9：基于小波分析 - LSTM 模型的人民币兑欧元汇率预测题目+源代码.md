# 案例 9：基于小波分析 - LSTM 模型的人民币兑欧元汇率预测
**问题背景**：汇率波动受宏观经济指标、国际贸易收支、利率政策、地缘政治等多种因素影响，短期汇率受投机交易等因素影响存在大量噪声，长期则呈现一定趋势。准确预测汇率对进出口企业规避汇率风险、制定经营策略具有重要意义。
**问题描述**：某进出口企业需要对人民币兑欧元的未来 30 天的日汇率中间价进行预测。要求模型能够分离汇率序列中的高频噪声（如短期投机交易影响）和低频有效信号（如宏观经济趋势），提高预测精度，帮助企业提前锁定汇率成本。
**数据情况**：提供过去 10 年的人民币兑欧元日汇率中间价数据，同时提供中国和欧元区的 GDP 增长率、通货膨胀率、利率、进出口贸易差额等宏观经济数据，以及重大国际事件（如贸易摩擦、政策调整）记录。数据存在个别交易日因节假日导致的缺失。
案例 9：小波分析 - LSTM 模型汇率预测代码

| import pandas as pd<br>import numpy as np<br>import matplotlib.pyplot as plt<br>import pywt<br>from sklearn.preprocessing import MinMaxScaler<br>from tensorflow.keras.models import Sequential<br>from tensorflow.keras.layers import LSTM, Dense<br>from sklearn.metrics import mean_squared_error<br># 数据加载与预处理<br>data = pd.read_csv('exchange_rate.csv', parse_dates=['date'], index_col='date')<br>rate_data = data['rate'].values.reshape(-1, 1)<br># 数据归一化<br>scaler = MinMaxScaler(feature_range=(0, 1))<br>rate_scaled = scaler.fit_transform(rate_data)<br># 小波分解<br>def wavelet_decompose(data, wavelet='db4', level=3):<br>coeffs = pywt.wavedec(data, wavelet, level=level)<br>return coeffs  # 返回近似系数和细节系数<br># 小波重构<br>def wavelet_reconstruct(coeffs, wavelet='db4'):<br>return pywt.waverec(coeffs, wavelet)<br># 分解序列<br>coeffs = wavelet_decompose(rate_scaled.flatten(), level=3)<br>n_levels = len(coeffs)<br># 为每个分量构建LSTM模型并预测<br>predictions = []<br>for i in range(n_levels):<br>component = coeffs[i].reshape(-1, 1)<br># 构建序列数据<br>def create_seq(data, look_back=10):<br>X, y = [], []<br>for j in range(len(data) - look_back - 1):<br>X.append(data[j:j+look_back, 0])<br>y.append(data[j+look_back, 0])<br>return np.array(X), np.array(y)<br>look_back = 10<br>X, y = create_seq(component, look_back)<br>X = X.reshape(X.shape[0], X.shape[1], 1)<br># 划分训练集和测试集<br>train_size = int(len(X) * 0.8)<br>X_train, X_test = X[:train_size], X[train_size:]<br>y_train, y_test = y[:train_size], y[train_size:]<br># 训练LSTM模型<br>model = Sequential()<br>model.add(LSTM(32, input_shape=(look_back, 1)))<br>model.add(Dense(1))<br>model.compile(loss='mse', optimizer='adam')<br>model.fit(X_train, y_train, epochs=20, batch_size=16, verbose=0)<br># 预测<br>y_pred = model.predict(X_test)<br>predictions.append(y_pred)<br># 小波重构预测结果<br># 对齐各分量长度<br>min_len = min(len(p) for p in predictions)<br>aligned_preds = [p[:min_len] for p in predictions]<br># 重构<br>pred_coeffs = []<br>for i in range(n_levels):<br># 低通分量（近似系数）需要补全长度<br>if i == 0:<br>pred_coeff = np.zeros_like(coeffs[i])<br>pred_coeff[-min_len:] = aligned_preds[i].flatten()<br>else:<br>pred_coeff = np.zeros_like(coeffs[i])<br>pred_coeff[-min_len:] = aligned_preds[i].flatten()<br>pred_coeffs.append(pred_coeff)<br># 重构序列<br>final_pred = wavelet_reconstruct(pred_coeffs)<br>final_pred = final_pred[-min_len:].reshape(-1, 1)<br># 反归一化<br>final_pred = scaler.inverse_transform(final_pred)<br>test_actual = scaler.inverse_transform(rate_scaled[-min_len:])<br># 评估模型<br>mse = mean_squared_error(test_actual, final_pred)<br>print(f'小波-LSTM模型MSE: {mse}')<br># 可视化结果<br>plt.figure(figsize=(12, 6))<br>plt.plot(test_actual, label='实际汇率')<br>plt.plot(final_pred, label='预测汇率')<br>plt.legend()<br>plt.savefig('exchange_rate_prediction.png')<br>plt.show() |
