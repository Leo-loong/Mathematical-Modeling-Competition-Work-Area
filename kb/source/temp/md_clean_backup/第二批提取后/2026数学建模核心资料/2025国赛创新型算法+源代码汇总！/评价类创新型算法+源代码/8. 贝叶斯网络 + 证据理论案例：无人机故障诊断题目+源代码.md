# 8. 贝叶斯网络 + 证据理论案例：无人机故障诊断
**问题背景**：某无人机厂商需构建故障诊断系统，融合 5 个传感器数据（电池电压、电机转速、GPS 信号强度等），判断无人机是否存在 “动力系统故障”，传感器数据存在噪声和冲突（如某传感器显示异常但其他正常）。
**数据**：
1000 组飞行记录，每组含 5 项传感器数据（数值型）及故障标签（1 = 故障，0 = 正常），部分记录存在传感器数据缺失。
**要求**：用贝叶斯网络建模传感器与故障的因果关系，用证据理论融合冲突数据，输出故障概率及诊断置信度，分析对缺失数据的处理能力。
### 8. 贝叶斯网络 + 证据理论代码：无人机故障诊断

| import numpy as np<br>import pandas as pd<br>from pgmpy.models import BayesianNetwork<br>from pgmpy.estimators import MaximumLikelihoodEstimator<br>from pgmpy.inference import VariableElimination<br>from scipy.stats import norm<br># 1. 数据准备（模拟1000组飞行记录）<br>np.random.seed(42)<br>n_samples = 1000<br># 传感器数据（5项：电池电压、电机转速、GPS信号、温度、振动）<br># 正常状态下的分布<br>battery_voltage = norm.rvs(loc=12.0, scale=0.5, size=n_samples)<br>motor_speed = norm.rvs(loc=3000, scale=200, size=n_samples)<br>gps_signal = norm.rvs(loc=80, scale=10, size=n_samples)  # 信号强度0-100<br>temperature = norm.rvs(loc=45, scale=5, size=n_samples)<br>vibration = norm.rvs(loc=0.1, scale=0.05, size=n_samples)<br># 故障标签（10%故障概率）<br>fault = np.random.binomial(1, 0.1, n_samples)<br># 故障状态下传感器数据异常<br>battery_voltage[fault == 1] = norm.rvs(loc=9.0, scale=1.0, size=sum(fault))<br>motor_speed[fault == 1] = norm.rvs(loc=2000, scale=300, size=sum(fault))<br>gps_signal[fault == 1] = norm.rvs(loc=40, scale=15, size=sum(fault))<br>temperature[fault == 1] = norm.rvs(loc=60, scale=8, size=sum(fault))<br>vibration[fault == 1] = norm.rvs(loc=0.5, scale=0.2, size=sum(fault))<br># 离散化传感器数据（3个状态：低、中、高）<br>def discretize(data, bins=3):<br>return np.digitize(data, np.percentile(data, [100*i/bins for i in range(1, bins)]))<br>sensors_discrete = np.column_stack([<br>discretize(battery_voltage),<br>discretize(motor_speed),<br>discretize(gps_signal),<br>discretize(temperature),<br>discretize(vibration)<br>])<br># 构造数据集<br>data = pd.DataFrame(sensors_discrete, columns=['电池电压', '电机转速', 'GPS信号', '温度', '振动'])<br>data['动力系统故障'] = fault<br># 2. 贝叶斯网络建模<br>model = BayesianNetwork([<br>('电池电压', '动力系统故障'),<br>('电机转速', '动力系统故障'),<br>('GPS信号', '动力系统故障'),<br>('温度', '动力系统故障'),<br>('振动', '动力系统故障')<br>])<br># 参数学习<br>model.fit(data, estimator=MaximumLikelihoodEstimator)<br># 3. 证据理论融合<br>def dempster_shafer(evidence_list):<br># 基本概率分配（BPA）<br>bpa = {0: 0.1, 1: 0.1}  # 初始不确定性<br>for evidence in evidence_list:<br>new_bpa = {0: 0, 1: 0}<br># 传感器证据支持度<br>p_fault = evidence['P(故障\|证据)']<br>new_bpa[1] += p_fault * 0.8<br>new_bpa[0] += (1 - p_fault) * 0.8<br># 剩余概率分配给不确定性<br>new_bpa[0] += 0.2 * bpa[0]<br>new_bpa[1] += 0.2 * bpa[1]<br>bpa = new_bpa<br># 归一化<br>total = sum(bpa.values())<br>return {k: v/total for k, v in bpa.items()}<br># 4. 故障诊断示例（随机选择10条记录）<br>infer = VariableElimination(model)<br>sample_indices = np.random.choice(n_samples, 10, replace=False)<br>results = []<br>for idx in sample_indices:<br>sample = data.iloc[idx]<br>evidence = {col: sample[col] for col in ['电池电压', '电机转速', 'GPS信号', '温度', '振动']}<br># 贝叶斯网络推理<br>fault_proba = infer.query(variables=['动力系统故障'], evidence=evidence)['动力系统故障'].values[1]<br># 证据融合<br>evidence_list = [{'P(故障\|证据)': fault_proba}]<br>fused_bpa = dempster_shafer(evidence_list)<br># 诊断结果<br>diagnosis = 1 if fused_bpa[1] > 0.5 else 0<br>results.append({<br>'样本索引': idx,<br>'实际故障': sample['动力系统故障'],<br>'贝叶斯故障概率': round(fault_proba, 4),<br>'融合后故障概率': round(fused_bpa[1], 4),<br>'诊断结果': diagnosis<br>})<br># 展示结果<br>result_df = pd.DataFrame(results)<br>print("无人机故障诊断结果：")<br>print(result_df)<br>print(f"\n诊断准确率：{sum(result_df['实际故障'] == result_df['诊断结果'])/len(result_df):.2%}") |
